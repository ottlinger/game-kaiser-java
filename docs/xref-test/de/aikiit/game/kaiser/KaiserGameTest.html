51      @Test
52      void tenActionYearsAreFollowedByEleventhReportAndThenEvaluation() {
53          BasicRandom random = tenYearRandom();
54          KaiserEngine engine = new KaiserEngine(random);
55          String output = run(engine, tenYearInput() + "n");
56          assertThat(engine.getState()).isEqualTo(KaiserEngine.GameState.COMPLETED);
57          assertThat(engine.getYear()).isEqualTo(11);
58          assertThat(engine.getRegency()).isEqualTo(KaiserEngine.Regency.MEDIOCRE);
59          assertThat(engine.getTotalStarvationDeaths()).isZero();
60          assertThat(engine.getPopulation()).isEqualByComparingTo("137");
61          assertThat(engine.getLandArea()).isEqualByComparingTo("1330");
62          assertThat(engine.getGrainStocks()).isEqualByComparingTo("26050");
63          assertThat(engine.getImmigrants()).isEqualByComparingTo("4");
64          assertThat(output.split("Wieviel Land wollen Sie kaufen", -1)).hasSize(11);
65          assertThat(output).contains("Im Jahr 11", "glücklicher", "Nochmal (J/N)?");
66          assertThat(output.indexOf("Im Jahr 11")).isLessThan(output.indexOf("Bewertung Ihrer Herrschaft"));
67          assertThat(output).doesNotContain("Im Jahr 0", "11-jährigen Amtszeit");
68          assertThat(random.remaining()).isEqualTo(1); // No land-price roll without a replay.
69      }

71      @Test
72      void finalReportStillAppliesImmigrationAndPlagueBeforeRating() {
73          List<Double> values = new ArrayList<>();
74          for (int year = 1; year <= 10; year++) {
75              values.addAll(List.of(0.0, 0.99, 0.0, 0.0, year == 10 ? 0.0 : 0.99));
76          }
77          BasicRandom random = new BasicRandom(values.toArray(Double[]::new));
78          KaiserEngine engine = new KaiserEngine(random);
79          String output = run(engine, tenYearInput());
80          assertThat(engine.getYear()).isEqualTo(11);
81          assertThat(engine.getPopulation()).isEqualByComparingTo("68"); // INT((133+4)/2).
82          assertThat(engine.getTotalStarvationDeaths()).isZero();
83          assertThat(engine.getRegency()).isEqualTo(KaiserEngine.Regency.EXCELLENT);
84          assertThat(output).contains("fürchterliche Seuche", "wunderbare Regierungszeit");
85          assertThat(output).doesNotContain("Nochmal");
86          assertThat(random.remaining()).isZero();

102     private static BasicRandom tenYearRandom() {
103         List<Double> values = new ArrayList<>();
104         for (int year = 1; year <= 10; year++) {
105             // Price 17, yield 5, no rats, factor 1, no plague.
106             values.addAll(List.of(0.0, 0.99, 0.0, 0.0, 0.99));
107         }
108         values.add(0.5); // Line 961: residents unhappy with the mediocre reign.
109         values.add(0.0); // Land price if the player chooses to play again.
110         return new BasicRandom(values.toArray(Double[]::new));
111     }

113     /** A solvent ten-year strategy; keep roughly ten hectares per person. */
114     private static String tenYearInput() {
115         StringBuilder input = new StringBuilder("j ");
116         long population = 100;
117         long area = 1000;
118         long stocks = 2800;
119         for (int year = 1; year <= 10; year++) {
120             long buy = population * 10 - area;
121             input.append(buy).append(' ');
122             if (buy == 0) {
123                 input.append("0 "); // Selling is offered only after buying zero.
124             }
125             stocks -= buy * 17;
126             area += buy;
127             long food = population * 20;
128             long planted = area - 1; // BASIC's worker limit is strict.
129             input.append(food).append(' ').append(planted).append(' ');
130             stocks = stocks - food - planted / 2 + planted * 5;
131             population += (20 * area + stocks) / population / 100 + 1;
132         }
133         return input.toString();
134     }

136     private static String run(KaiserEngine engine, String input) {
137         ByteArrayOutputStream bytes = new ByteArrayOutputStream();
138         PrintStream original = System.out;
139         try (PrintStream output = new PrintStream(bytes, true, StandardCharsets.UTF_8)) {
140             System.setOut(output);
141             new KaiserGame(engine, new Scanner(input)).run();
142         } finally {
143             System.setOut(original);
144         }
145         return bytes.toString(StandardCharsets.UTF_8);
146     }
