3   import org.junit.jupiter.api.Test;
4   import org.junit.jupiter.params.ParameterizedTest;
5   import org.junit.jupiter.params.provider.CsvSource;

7   import static org.assertj.core.api.Assertions.assertThat;

9   class KaiserEnginePrinterTest {
10      @Test
11      void reportDisplaysPreviousHarvestRatherThanLandPrice() {
12          KaiserEngine engine = new KaiserEngine(new BasicRandom(0.0));
13          engine.startNewRound();
14          engine.prepareLandTrading();
15          KaiserEnginePrinter printer = new KaiserEnginePrinter(engine);
16          assertThat(printer.getStatus(1))
17                  .contains("3", "Die Ernte betrug (dt/Hektar)", "Ratten haben gefressen (dt)", "Eure Vorräte betragen (dt)")
18                  .doesNotContain("17", "DZ", "dzt");
19          assertThat(printer.getYearResult(1)).contains("Im Jahr 1", "5");
20          assertThat(printer.getResults()).contains("10-jährigen", "10 Hektar");
21      }
22  
23      @ParameterizedTest
24      @CsvSource({"0,EXCELLENT", "3,EXCELLENT", "4,MEDIOCRE", "10,MEDIOCRE", "11,TYRANNICAL",
25              "33,TYRANNICAL", "34,DISASTROUS"})
26      void starvationRatingThresholdsMatchBasic(int deaths, KaiserEngine.Regency rating) {
27          KaiserEngine engine = new KaiserEngine(new BasicRandom(0.0, 0.0, 0.0, 0.5));
28          engine.startNewRound();
29          engine.feedToPopulation(2000L - 20L * deaths);
30          engine.cultivate(0L);
31          engine.finishRoundAfterActions();
32          assertThat(engine.getRegency()).isEqualTo(rating);
33      }
34  
35      @ParameterizedTest
36      @CsvSource({"699,DISASTROUS", "700,TYRANNICAL", "899,TYRANNICAL", "900,MEDIOCRE",
37              "999,MEDIOCRE", "1000,EXCELLENT"})
38      void landRatingUsesUnroundedRatios(int area, KaiserEngine.Regency rating) {
39          KaiserEngine engine = new KaiserEngine(new BasicRandom(0.0));
40          engine.startNewRound();
41          engine.prepareLandTrading();
42          engine.sellLand(1000L - area);
43          assertThat(engine.getRegency()).isEqualTo(rating);
44      }
45  
46      @ParameterizedTest
47      @CsvSource({"0,wunderbare", "4,glücklicher", "11,hartherzige", "34,Misswirtschaft"})
48      void allFourEvaluationMessagesAreReachable(int deaths, String message) {
49          BasicRandom random = new BasicRandom(0.0, 0.0, 0.0, 0.5, 0.5);
50          KaiserEngine engine = new KaiserEngine(random);
51          engine.startNewRound();
52          engine.feedToPopulation(2000L - 20L * deaths);
53          engine.cultivate(0L);
54          engine.finishRoundAfterActions();
55          String evaluation = new KaiserEnginePrinter(engine).evaluateRegency();
56          assertThat(evaluation).contains(message);
57          if (deaths == 4) {
58              // INT(96 * .8 * .5), not a fixed 80% of the population.
59              assertThat(evaluation).contains("38 Einwohner");
60              assertThat(random.remaining()).isZero();
61          } else {
62              assertThat(random.remaining()).isEqualTo(1);
63          }
64      }
65  }
