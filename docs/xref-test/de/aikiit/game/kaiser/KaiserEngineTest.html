38      @Test
39      void feedingAndImmigrationFollowBasicTraceAndLandPersists() {
40          // Price 17, yield 3, rat roll 3 (no damage), immigration factor 1, no plague.
41          BasicRandom random = new BasicRandom(0.0, 0.4, 0.4, 0.0, 0.5);
42          KaiserEngine engine = new KaiserEngine(random);
43          engine.startNewRound();
44          engine.prepareLandTrading();
45          engine.buyLand(10L);
46          engine.feedToPopulation(1800L);
47          engine.cultivate(900L);
48          engine.finishRoundAfterActions();
49          // S=2800-170-1800-450+2700=3080; I=INT((20*1010+3080)/100/100+1)=3.
50          assertThat(engine.getGrainStocks()).isEqualByComparingTo("3080");
51          assertThat(engine.getImmigrants()).isEqualByComparingTo("3");
52          assertThat(engine.getPopulation()).isEqualByComparingTo("90");
53          assertThat(engine.getStarvationDeaths()).isEqualByComparingTo("10");
54          assertThat(engine.getTotalStarvationDeaths()).isEqualByComparingTo("10");
55          assertThat(engine.getAverageStarvationPercentage()).isEqualByComparingTo("10");
56          engine.startNewRound();
57          assertThat(engine.getPopulation()).isEqualByComparingTo("93");
58          assertThat(engine.getLandArea()).isEqualByComparingTo("1010");
59          assertThat(engine.getRatLoss()).isZero();
60          assertThat(engine.getHarvestYield()).isEqualByComparingTo("3");
61          assertThat(random.remaining()).isZero();
62      }

66      void plagueIsRolledAtYearEndAndAppliedIncludingZero(double roll, int plagueRoll, boolean plague) {
67          KaiserEngine engine = new KaiserEngine(new BasicRandom(0.0, 0.0, 0.0, roll));
68          engine.startNewRound();
69          engine.feedToPopulation(2000L);
70          engine.cultivate(0L);
71          engine.finishRoundAfterActions();
72          // S=800, I=3: next year's population is 103, or INT(103/2)=51.
73          assertThat(engine.getPlagueRoll()).isEqualTo(plagueRoll);
74          engine.startNewRound();
75          assertThat(engine.isPlagueOccurred()).isEqualTo(plague);
76          assertThat(engine.getPopulation()).isEqualByComparingTo(plague ? "51" : "103");
77          assertThat(engine.getTotalStarvationDeaths()).isZero();
78      }

92      @ParameterizedTest
93      @CsvSource({"1100,RUNNING,45", "1080,DISMISSED,46", "0,DISMISSED,100"})
94      void dismissalRequiresMoreThanFortyFivePercent(long food, KaiserEngine.GameState state, String deaths) {
95          KaiserEngine engine = new KaiserEngine(new BasicRandom(0.0, 0.0, 0.0, 0.5));
96          engine.startNewRound();
97          engine.feedToPopulation(food);
98          engine.cultivate(0L);
99          engine.finishRoundAfterActions();
100         assertThat(engine.getState()).isEqualTo(state);
101         assertThat(engine.getStarvationDeaths()).isEqualByComparingTo(deaths);
102         if (state == KaiserEngine.GameState.DISMISSED) {
103             assertThat(engine.getTotalStarvationDeaths()).isZero();
104             assertThatThrownBy(engine::startNewRound)
105                     .isInstanceOf(IllegalStateException.class)
106                     .hasMessage("The game has ended.");
107         }
108     }

110     @Test
111     void deathPercentagesKeepFractionsAcrossYears() {
112         KaiserEngine engine = new KaiserEngine(new BasicRandom(
113                 0.99, 0.4, 0.0, 0.5, 0.99, 0.4, 0.0, 0.5));
114         engine.startNewRound();
115         engine.feedToPopulation(1980L); // 1/100 dies; next population = 99+3=102.
116         engine.cultivate(900L);
117         engine.finishRoundAfterActions();
118         engine.startNewRound();
119         engine.feedToPopulation(2020L); // 1/102 dies.
120         engine.cultivate(900L);
121         engine.finishRoundAfterActions();
122         assertThat(engine.getTotalStarvationDeaths()).isEqualByComparingTo("2");
123         assertThat(engine.getAverageStarvationPercentage()).isBetween(new BigDecimal("0.99019"), new BigDecimal("0.99020"));

139     void exactFeedingUpdatesAverageButOverfeedingLeavesItUntouched(long food, String average) {
140         KaiserEngine engine = new KaiserEngine(new BasicRandom(

142         engine.startNewRound();

145         engine.finishRoundAfterActions();
146         assertThat(engine.getAverageStarvationPercentage()).isEqualByComparingTo("10");
147         engine.startNewRound();
148         engine.feedToPopulation(food); // Population 93: exactly 1860 grain is needed.
149         engine.cultivate(0L);
150         engine.finishRoundAfterActions();
151         assertThat(engine.getPopulation()).isEqualByComparingTo("93");
152         assertThat(engine.getStarvationDeaths()).isZero();
153         assertThat(engine.getAverageStarvationPercentage()).isEqualByComparingTo(average);
154         assertThat(engine.getTotalStarvationDeaths()).isEqualByComparingTo("10");

163     @Test
164     void resetRestoresAllInitialState() {
165         KaiserEngine engine = new KaiserEngine(new BasicRandom(0.0));
166         engine.startNewRound();
167         engine.prepareLandTrading();
168         engine.buyLand(1L);
169         engine.abort();
170         engine.reset();
171         assertThat(engine.getState()).isEqualTo(KaiserEngine.GameState.RUNNING);
172         assertThat(engine.getYear()).isZero();
173         assertThat(engine.getPopulation()).isEqualByComparingTo("95");
174         assertThat(engine.getLandArea()).isEqualByComparingTo("1000");
175         assertThat(engine.getGrainStocks()).isEqualByComparingTo("2800");
176         assertThat(engine.getLandPrice()).isZero();
