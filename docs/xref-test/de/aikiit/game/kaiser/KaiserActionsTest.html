5   import java.util.Scanner;

7   import static org.assertj.core.api.Assertions.assertThat;

9   class KaiserActionsTest {
10      @Test
11      void sharedScannerPreservesAllFourActionsAndRetriesBadInput() {
12          KaiserEngine engine = new KaiserEngine(new BasicRandom(0.0, 0.99, 0.4, 0.0));
13          engine.startNewRound();
14          engine.prepareLandTrading();
15          Scanner input = new Scanner("abc 99999999999999999999999999999 999999 0 1000 5 9999 2000 1001 900");
16          assertThat(KaiserActions.buy(engine, input)).isFalse();
17          KaiserActions.sell(engine, input);
18          KaiserActions.feed(engine, input);
19          KaiserActions.cultivate(engine, input);
20          assertThat(engine.getLandArea()).isEqualByComparingTo("995");
21          assertThat(engine.getFoodAllocated()).isEqualByComparingTo("2000");
22          assertThat(engine.getTotalHarvest()).isEqualByComparingTo("4500");
23          assertThat(input.hasNext()).isFalse();
24      }
25  
26      @Test
27      void basicDecimalInputsAreAcceptedAndIntOperationsStillFloor() {
28          KaiserEngine engine = new KaiserEngine(new BasicRandom(0.0, 0.4, 0.4, 0.0));
29          engine.startNewRound();
30          engine.prepareLandTrading();
31          Scanner input = new Scanner("1.5 1999.9 3.9");
32          assertThat(KaiserActions.buy(engine, input)).isTrue();
33          KaiserActions.feed(engine, input);
34          KaiserActions.cultivate(engine, input);
35          assertThat(engine.getLandArea()).isEqualByComparingTo("1001.5");
36          assertThat(engine.getFoodAllocated()).isEqualByComparingTo("1999.9");
37          assertThat(engine.getTotalHarvest()).isEqualByComparingTo("11.7");
38          // S=2800-25.5-1999.9-INT(3.9/2)+11.7.
39          assertThat(engine.getGrainStocks()).isEqualByComparingTo("785.3");
40      }
41  
42      @Test
43      void positivePurchaseSkipsSelling() {
44          KaiserEngine engine = new KaiserEngine(new BasicRandom(0.0));
45          engine.startNewRound();
46          engine.prepareLandTrading();
47          assertThat(KaiserActions.buy(engine, new Scanner("11"))).isTrue();
48          assertThat(engine.getLandArea()).isEqualByComparingTo("1011");
49      }
50  
51      @Test
52      void negativeInputAndEndOfInputAbortInsteadOfRetryingForever() {
53          KaiserEngine negative = new KaiserEngine(new BasicRandom());
54          KaiserActions.feed(negative, new Scanner("-1"));
55          assertThat(negative.getState()).isEqualTo(KaiserEngine.GameState.ABORTED);
56          KaiserEngine eof = new KaiserEngine(new BasicRandom());
57          KaiserActions.buy(eof, new Scanner(""));
58          assertThat(eof.getState()).isEqualTo(KaiserEngine.GameState.ABORTED);
59      }
