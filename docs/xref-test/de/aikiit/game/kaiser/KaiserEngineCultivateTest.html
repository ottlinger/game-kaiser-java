3   import org.junit.jupiter.api.Test;
4   import org.junit.jupiter.params.ParameterizedTest;
5   import org.junit.jupiter.params.provider.CsvSource;
6   
7   import static org.assertj.core.api.Assertions.assertThat;
8   import static org.assertj.core.api.Assertions.assertThatThrownBy;

10  class KaiserEngineCultivateTest {
11      @Test
12      void zeroStillCalculatesHarvestRatsAndImmigrationFactor() {
13          BasicRandom random = new BasicRandom(0.99, 0.2, 0.99);
14          KaiserEngine engine = new KaiserEngine(random);
15          engine.startNewRound();
16          engine.cultivate(0L);
17          assertThat(engine.getHarvestYield()).isEqualByComparingTo("5");
18          assertThat(engine.getTotalHarvest()).isZero();
19          assertThat(engine.getRatLoss()).isEqualByComparingTo("1400");
20          assertThat(engine.getGrainStocks()).isEqualByComparingTo("1400");
21          assertThat(engine.getImmigrationFactor()).isEqualTo(5);
22          assertThat(random.remaining()).isZero();
23      }
24  
25      @ParameterizedTest
26      @CsvSource({"0,0", "0.2,1400", "0.4,0", "0.6,700", "0.8,0"})
27      void onlyEvenRatRollsCauseDamage(double roll, String loss) {
28          KaiserEngine engine = new KaiserEngine(new BasicRandom(0.0, roll, 0.0));
29          engine.cultivate(0L);
30          assertThat(engine.getRatLoss()).isEqualByComparingTo(loss);
31      }
32  
33      @Test
34      void seedAndRatLossUseFloorNotRounding() {
35          KaiserEngine engine = new KaiserEngine(new BasicRandom(0.0, 0.2, 0.0));
36          engine.startNewRound();
37          engine.cultivate(3L);
38          // S=2800-INT(3/2)=2799; E=INT(2799/2)=1399; H=3.
39          assertThat(engine.getRatLoss()).isEqualByComparingTo("1399");
40          assertThat(engine.getGrainStocks()).isEqualByComparingTo("1403");
41      }
42  
43      @Test
44      void exactSeedStockIsRejectedButZeroPlantingBypassesValidation() {
45          KaiserEngine engine = new KaiserEngine(new BasicRandom(0.0, 0.0, 0.0));
46          engine.startNewRound();
47          engine.feedToPopulation(2798L);
48          assertThatThrownBy(() -> engine.cultivate(4L))
49                  .isInstanceOf(IllegalArgumentException.class)
50                  .hasMessage("You cannot cultivate more than you have.");
51          assertThat(engine.getGrainStocks()).isEqualByComparingTo("2");
52          engine.cultivate(0L);
53      }
54  
55      @Test
56      void exactWorkerCapacityIsRejectedButPlantingAllLandIsAllowed() {
57          KaiserEngine engine = new KaiserEngine(new BasicRandom(0.0, 0.0, 0.0));
58          // Initial P=95, so 950 is forbidden; 949 is allowed.
59          assertThatThrownBy(() -> engine.cultivate(950L))
60                  .isInstanceOf(IllegalArgumentException.class)
61                  .hasMessage("Not enough workers available.");
62          engine.cultivate(949L);
63  
64          KaiserEngine allLand = new KaiserEngine(new BasicRandom(0.99, 0.4, 0.0, 0.5,
65                  0.0, 0.0, 0.0));
66          allLand.startNewRound();
67          allLand.feedToPopulation(2000L);
68          allLand.cultivate(900L);
69          allLand.finishRoundAfterActions();
70          allLand.startNewRound(); // P=103: 1000 hectares now strictly below capacity.
71          allLand.cultivate(1000L);
72          assertThat(allLand.getTotalHarvest()).isEqualByComparingTo("1000");
73      }
74  
75      @Test
76      void moreLandThanOwnedIsRejectedWithoutRandomDraws() {
77          KaiserEngine engine = new KaiserEngine(new BasicRandom());
78          assertThatThrownBy(() -> engine.cultivate(1001L))
79                  .isInstanceOf(IllegalArgumentException.class)
80                  .hasMessage("You cannot cultivate more area than you have.");
81          assertThat(engine.getGrainStocks()).isEqualByComparingTo("2800");
82      }
83  
84      @Test
85      void negativeInputEndsGameWithoutHarvest() {
86          KaiserEngine engine = new KaiserEngine(new BasicRandom());
87          engine.cultivate(-1L);
88          assertThat(engine.getState()).isEqualTo(KaiserEngine.GameState.ABORTED);
89          assertThat(engine.getGrainStocks()).isEqualByComparingTo("2800");
90      }
